import { useEffect, useState } from "react";

import GoalModal from "./components/GoalModal";
import Profile from "./pages/Profile";

import {
  getScheduledBills,
  createScheduledBill,
  updateScheduledBill,
  deleteScheduledBill,
} from "./services/scheduledBillService";

import {
  getGoal,
  createGoal,
  updateGoal,
} from "./services/goalService";

import {
  getLoans,
  getLoanById,
  createLoan,
  deleteLoan,
  markLoanAsPaid,
  updateLoan,
} from "./services/loanService";

import {
  getTransactions,
  deleteTransaction,
  getPreservedFinancialBalances,
  getPreservedChartHistory,
  cleanTransactionHistory,
  countTransactionsToClean,
} from "./services/transactionService";

import {
  getUserSettings,
  saveSpendingLimit,
} from "./services/userSettingsService";

import LoanList from "./components/LoanList";
import LoanModal from "./components/LoanModal";
import Header from "./components/Header";
import TransactionList from "./components/TransactionList";
import Modal from "./components/Modal";
import ModalWizard from "./components/modal/ModalWizard";
import Home from "./components/Home";
import FilterBar from "./components/FilterBar";
import Login from "./components/auth/Login";
import ScheduledBillModal from "./components/ScheduledBillModal";

import "./styles/dashboard.css";

import { supabase } from "./services/supabase";

function App() {
  const [transactions, setTransactions] = useState([]);
  const [saldoBase, setSaldoBase] = useState(0);
  const [investimentosBase, setInvestimentosBase] = useState(0);
  const [financialDataLoaded, setFinancialDataLoaded] = useState(false);
  const [cleaningHistory, setCleaningHistory] = useState(false);
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [editingTransaction, setEditingTransaction] = useState(null);

  const [cleanupPeriod, setCleanupPeriod] = useState("3");
  const [cleanupCount, setCleanupCount] = useState(null);
  const [countingCleanup, setCountingCleanup] = useState(false);
  const [cleanupCountError, setCleanupCountError] = useState("");

  const [showFilters, setShowFilters] = useState(false);
  const [filtro, setFiltro] = useState("Todos");

  const [user, setUser] = useState(null);
  const [loadingAuth, setLoadingAuth] = useState(true);

  const [loans, setLoans] = useState([]);
  const [loanModalOpen, setLoanModalOpen] = useState(false);
  const [editingLoan, setEditingLoan] = useState(null);

  const [goal, setGoal] = useState(null);
  const [goalModalOpen, setGoalModalOpen] = useState(false);
  const [editingGoal, setEditingGoal] = useState(null);

  const [scheduledBills, setScheduledBills] = useState([]);
  const [scheduledBillModalOpen, setScheduledBillModalOpen] = useState(false);
  const [editingScheduledBill, setEditingScheduledBill] = useState(null);

  const [showLoans, setShowLoans] = useState(true);
  const [showScheduledBills, setShowScheduledBills] = useState(true);
  const [showTransactions, setShowTransactions] = useState(true);

  const [limiteGastos, setLimiteGastos] = useState(3000); 
  const [valorGuardadoUtilizadoBase, setValorGuardadoUtilizadoBase] = useState(0);

  const [preservedChartHistory, setPreservedChartHistory] = useState([]);


  
  async function loadTransactions() {
    if (!user) return false;
  
    setFinancialDataLoaded(false);
  
    try {
      const [
        transactionsData,
        balances,
        chartHistoryData,
      ] = await Promise.all([
        getTransactions(user.id),
        getPreservedFinancialBalances(user.id),
        getPreservedChartHistory(user.id),
      ]);
  
      setTransactions(transactionsData);
      setSaldoBase(balances.saldoBase);
      setInvestimentosBase(balances.investimentosBase);
      setValorGuardadoUtilizadoBase(
        Number(balances?.valorGuardadoUtilizadoBase ?? 0)
      );
  
      setPreservedChartHistory(chartHistoryData);
  
      setFinancialDataLoaded(true);
  
      return true;
    } catch (error) {
      console.error("Erro ao carregar dados financeiros:", error);
  
      window.alert(
        "Não foi possível carregar seus dados financeiros. Verifique sua conexão e tente novamente."
      );
  
      return false;
    }
  }


  // Controla a navegação entre o dashboard e o perfil.
  const [currentPage, setCurrentPage] = useState("home");

  async function removeLoan(id) {
    if (!confirm("Excluir empréstimo?")) return;

    await deleteLoan(id);
    await loadLoans();
  }

  async function payLoan(id) {
    const loan = await getLoanById(id);

    await supabase.from("transactions").insert({
      user_id: user.id,
      tipo: loan.tipo === "receber" ? "entrada" : "saida",
      categoria: "Empréstimos",
      descricao:
        loan.tipo === "receber"
          ? `Recebido de ${loan.pessoa}`
          : `Pagamento para ${loan.pessoa}`,
      valor: Number(loan.valor),
      data: new Date().toISOString().slice(0, 10),
      observacao: loan.descricao,
    });

    await markLoanAsPaid(id);

    await loadTransactions();
    await loadLoans();
  }

  async function saveLoan(loan) {
    if (loan.id) {
      await updateLoan(loan.id, loan);
    } else {
      await createLoan({
        ...loan,
        valor: Number(loan.valor),
        status: loan.status,
        user_id: user.id,
      });
    }

    await loadLoans();

    setEditingLoan(null);
    setLoanModalOpen(false);
  }

  

 
async function handleUseSavedMoney(valor) {
  if (!user) {
    throw new Error("Você precisa estar autenticado.");
  }

  if (!goal) {
    throw new Error("Cadastre uma meta antes de usar o valor guardado.");
  }

  const valorUtilizado = Number(valor);

  if (!Number.isFinite(valorUtilizado) || valorUtilizado <= 0) {
    throw new Error("Informe um valor válido.");
  }

  
/* ALTERAÇÃO: calcular o total já utilizado */

const totalJaUtilizado =
  Number(valorGuardadoUtilizadoBase || 0) +
  transactions
    .filter(
      (item) =>
        item.tipo === "saida" &&
        item.descricao === "Valor guardado usado"
    )
    .reduce(
      (total, item) => total + Number(item.valor || 0),
      0
    );


/* ALTERAÇÃO: considerar o saldo inicial de investimentos
   e os investimentos registrados no histórico */
const totalInvestimentosGuardados =
  Number(investimentosBase || 0) +
  transactions
    .filter((item) => item.tipo === "investimento")
    .reduce(
      (total, item) => total + Number(item.valor || 0),
      0
    );

/* ALTERAÇÃO: usar a mesma regra de cálculo do Dashboard */
const valorDisponivel = Math.max(
  0,
  Number(goal.valor_inicial || 0) +
    totalInvestimentosGuardados -
    totalJaUtilizado
);

/* ALTERAÇÃO: validar contra o saldo realmente disponível */
if (valorUtilizado > valorDisponivel + 0.001) {
  throw new Error(
    "O valor informado ultrapassa o dinheiro guardado disponível."
  );
}


  const hoje = new Date();
  const dataLocal = [
    hoje.getFullYear(),
    String(hoje.getMonth() + 1).padStart(2, "0"),
    String(hoje.getDate()).padStart(2, "0"),
  ].join("-");

  const { error } = await supabase
    .from("transactions")
    .insert({
      user_id: user.id,
      tipo: "saida",
      categoria: "Valor guardado",
      descricao: "Valor guardado usado",
      valor: valorUtilizado,
      data: dataLocal,
      observacao: "Utilização de dinheiro guardado.",
    });

  if (error) {
    console.error("Erro ao utilizar valor guardado:", error);
    throw new Error("Não foi possível registrar a utilização no Supabase.");
  }

  await loadTransactions();
}

  

  useEffect(() => {
    if (!user || !financialDataLoaded || cleaningHistory) {
      setCleanupCount(null);
      return;
    }
  
    let cancelled = false;
  
    async function updateCleanupCount() {
      setCountingCleanup(true);
      setCleanupCount(null);
      setCleanupCountError("");
  
      try {
        const count = await countTransactionsToClean(
          user.id,
          cleanupPeriod === "todos"
            ? "todos"
            : Number(cleanupPeriod)
        );
  
        if (!cancelled) {
          setCleanupCount(count);
        }
      } catch (error) {
        console.error("Erro ao contar transações:", error);
  
        if (!cancelled) {
          setCleanupCountError(
            "Não foi possível calcular a quantidade. Tente novamente."
          );
        }
      } finally {
        if (!cancelled) {
          setCountingCleanup(false);
        }
      }
    }
  
    updateCleanupCount();
  
    return () => {
      cancelled = true;
    };
  }, [
    user?.id,
    financialDataLoaded,
    cleanupPeriod,
    cleaningHistory,
  ]);

  async function loadLoans() {
    if (!user) return;

    const data = await getLoans(user.id);
    setLoans(data);
  }

  async function loadGoal() {
    if (!user) return;

    const data = await getGoal(user.id);
    setGoal(data);
  }

  async function loadUserSettings() {
    if (!user) return;

    try {
      const limite = await getUserSettings(user.id);
      setLimiteGastos(limite);
    } catch (error) {
      console.error("Erro ao carregar configurações:", error);
    }
  }

  async function handleSaveSpendingLimit(novoLimite) {
    if (!user) {
      throw new Error("Usuário não autenticado.");
    }

    const limiteSalvo = await saveSpendingLimit(
      user.id,
      novoLimite
    );

    setLimiteGastos(limiteSalvo);
  }

  async function loadScheduledBills() {
    if (!user) return;

    try {
      const data = await getScheduledBills(user.id);
      setScheduledBills(data);
    } catch (error) {
      console.error("ERRO AO CARREGAR CONTAS:", error);
    }
  }

  async function saveScheduledBill(bill) {
    if (bill.id) {
      await updateScheduledBill(bill.id, {
        nome: bill.nome,
        categoria: bill.categoria,
        valor: Number(bill.valor),
        tipo: bill.tipo,
        vencimento: bill.vencimento,
        recorrencia: bill.recorrencia,
        dias_lembrete: Number(bill.dias_lembrete),
        ativo: bill.ativo,
      });
    } else {
      await createScheduledBill({
        nome: bill.nome,
        categoria: bill.categoria,
        valor: Number(bill.valor),
        tipo: bill.tipo,
        vencimento: bill.vencimento,
        recorrencia: bill.recorrencia,
        dias_lembrete: Number(bill.dias_lembrete),
        ativo: bill.ativo,
        user_id: user.id,
      });
    }

    await loadScheduledBills();

    setScheduledBillModalOpen(false);
    setEditingScheduledBill(null);
  }

  async function removeScheduledBill(id) {
    if (!confirm("Excluir esta conta agendada?")) return;

    try {
      await deleteScheduledBill(id);
      await loadScheduledBills();
    } catch (error) {
      console.error("ERRO AO EXCLUIR CONTA:", error);
    }
  }

 
async function saveGoal(goalData) {
  if (!user) {
    throw new Error("Usuário não autenticado.");
  }

  const dadosMeta = {
    nome: "Investimentos/Guardados",
    valor_meta: Number(goalData.valor_meta),
    valor_inicial: Number(goalData.valor_inicial || 0),
    prazo_meses: Number(goalData.prazo_meses),
  };

  if (
    !Number.isFinite(dadosMeta.valor_meta) ||
    dadosMeta.valor_meta <= 0
  ) {
    throw new Error("O valor-alvo precisa ser maior que zero.");
  }

  if (
    !Number.isFinite(dadosMeta.valor_inicial) ||
    dadosMeta.valor_inicial < 0
  ) {
    throw new Error("O valor inicial não pode ser negativo.");
  }

  if (
    !Number.isInteger(dadosMeta.prazo_meses) ||
    dadosMeta.prazo_meses <= 0
  ) {
    throw new Error("O prazo precisa ser um número inteiro de meses.");
  }

  if (goalData.id) {
    await updateGoal(goalData.id, dadosMeta);
  } else {
    await createGoal({
      ...dadosMeta,
      user_id: user.id,
    });
  }

  await loadGoal();

  setGoalModalOpen(false);
  setEditingGoal(null);
}


  function handleEditGoalFromDashboard() {
    setEditingGoal(goal);
    setGoalModalOpen(true);
  }

  // Verifica a sessão e acompanha o estado de autenticação.
  useEffect(() => {
    supabase.auth.getSession().then(({ data }) => {
      setUser(data.session?.user ?? null);
      setLoadingAuth(false);
    });

    const { data: listener } = supabase.auth.onAuthStateChange(
      (event, session) => {
        setUser(session?.user ?? null);
      }
    );

    return () => listener.subscription.unsubscribe();
  }, []);

  // Carrega os dados financeiros do usuário autenticado.
  useEffect(() => {
    if (user) {
      loadTransactions();
      loadLoans();
      loadGoal();
      loadScheduledBills();
      loadUserSettings();
    }
  }, [user]);

  async function removeTransaction(id) {
    try {
      await deleteTransaction(id);
      await loadTransactions();
    } catch (err) {
      console.error(err);
    }
  }

  

  async function handleCleanTransactionHistory(months = null) {
    if (!user) {
      window.alert("Você precisa estar autenticado.");
      return;
    }
  
    if (cleaningHistory || !financialDataLoaded) {
      window.alert("Aguarde o carregamento dos dados financeiros.");
      return;
    }
  
    if (
      months !== null &&
      (!Number.isInteger(months) || months < 1)
    ) {
      window.alert("Selecione um período válido.");
      return;
    }
  
    const confirmation =
      months === null
        ? "Todo o histórico será apagado. O saldo e os investimentos acumulados serão preservados. Deseja continuar?"
        : `Serão apagadas as transações anteriores ao limite de ${months} mês(es). O saldo e os investimentos acumulados serão preservados. Deseja continuar?`;
  
    if (!window.confirm(confirmation)) return;
  
    setCleaningHistory(true);
  
    try {
      const result = await cleanTransactionHistory(months);

      console.log("Resultado da limpeza:", result);
  
      // A exclusão já ocorreu no banco. Agora atualizamos a tela.
      const transactionsReloaded = await loadTransactions();
  
      await loadGoal();
  
      if (!transactionsReloaded) {
        window.alert(
          `A limpeza foi executada: ${result?.deleted_count ?? 0} transação(ões) removida(s). Porém, não foi possível atualizar os dados na tela. Recarregue o aplicativo.`
        );
        return result;
      }
  
      window.alert(
        `Limpeza concluída! ${result?.deleted_count ?? 0} transação(ões) removida(s).`
      );
  
      return result;
    } catch (error) {
      console.error("Erro ao limpar histórico:", error);
  
      window.alert(
        "Não foi possível confirmar a limpeza. Confira os dados no Supabase antes de tentar novamente."
      );
  
      return null;
    } finally {
      setCleaningHistory(false);
    }
  }
  


  function editTransaction(transaction) {
    setEditingTransaction(transaction);
    setIsModalOpen(true);
  }

  const filteredTransactions =
    filtro === "Todos"
      ? transactions
      : transactions.filter((item) => {
          if (filtro === "Entradas") {
            return item.tipo === "entrada";
          }

          if (filtro === "Saídas") {
            return item.tipo === "saida";
          }

          if (filtro === "Investimentos") {
            return item.tipo === "investimento";
          }

          if (filtro === "Renda Passiva") {
            return item.tipo === "renda_passiva";
          }

          return true;
        });

  if (loadingAuth) {
    return <h2>Carregando...</h2>;
  }

  if (!user) {
    return <Login />;
  }

  const totalInvestimentos =
  Number(investimentosBase || 0) +
  transactions
    .filter((item) => item.tipo === "investimento")
    .reduce(
      (total, item) => total + Number(item.valor || 0),
      0
    );


    
    const totalValorGuardadoUtilizado =
    Number(valorGuardadoUtilizadoBase || 0) +
    transactions
      .filter(
        (item) =>
          item.tipo === "saida" &&
          item.descricao === "Valor guardado usado"
      )
      .reduce(
        (total, item) => total + Number(item.valor || 0),
        0
      );

  
  const valorAtualMeta = goal
    ? Math.max(
        0,
        Number(goal.valor_inicial || 0) +
          totalInvestimentos -
          totalValorGuardadoUtilizado
      )
    : 0;
  
  const percentualMeta =
    goal && Number(goal.valor_meta) > 0
      ? Math.min(
          (valorAtualMeta / Number(goal.valor_meta)) * 100,
          100
        )
      : 0;
  
  /* ALTERAÇÃO: removida a segunda declaração duplicada de percentualMeta */
  
  return (
    <>
      {/* CABEÇALHO: permanece visível nas duas telas */}
      <Header
        currentPage={currentPage}
        onOpenProfile={() => setCurrentPage("profile")}
        onGoHome={() => setCurrentPage("home")}
      />
  
      {/* NAVEGAÇÃO: mostra o perfil ou o dashboard */}
      {currentPage === "profile" ? (
        <Profile />
      ) : (
        <>
          {/* Modal Nova Transação / Edição */}
          <Modal
            isOpen={isModalOpen}
            onClose={() => {
              setIsModalOpen(false);
              setEditingTransaction(null);
            }}
          >
            <ModalWizard
              user={user}
              transactions={transactions}
              setTransactions={setTransactions}
              onClose={() => {
                setIsModalOpen(false);
                setEditingTransaction(null);
                loadTransactions();
              }}
            />
          </Modal>

          {/* Modal Empréstimo */}
          <Modal
            isOpen={loanModalOpen}
            onClose={() => {
              setLoanModalOpen(false);
              setEditingLoan(null);
            }}
          >
            <LoanModal
              loan={editingLoan}
              onClose={() => {
                setLoanModalOpen(false);
                setEditingLoan(null);
              }}
              onSave={saveLoan}
            />
          </Modal>

          {/* Modal Meta */}
          <Modal
            isOpen={goalModalOpen}
            onClose={() => {
              setGoalModalOpen(false);
              setEditingGoal(null);
            }}
          >
            <GoalModal
              goal={editingGoal}
              onClose={() => {
                setGoalModalOpen(false);
                setEditingGoal(null);
              }}
              onSave={saveGoal}
            />
          </Modal>

          {/* Modal Contas Agendadas */}
          <Modal
            isOpen={scheduledBillModalOpen}
            onClose={() => {
              setScheduledBillModalOpen(false);
              setEditingScheduledBill(null);
            }}
          >
            <ScheduledBillModal
              bill={editingScheduledBill}
              onClose={() => {
                setScheduledBillModalOpen(false);
                setEditingScheduledBill(null);
              }}
              onSave={saveScheduledBill}
            />
          </Modal>

          {/* Ações */}
          <div className="actions">
            <button
              className="new-transaction-btn"
              onClick={() => {
                setEditingTransaction(null);
                setIsModalOpen(true);
              }}
            >
              ➕ Nova Transação
            </button>

            <details className="history-cleanup panel-card">
  <summary className="history-cleanup-summary">
    <span>🗂️ Gerenciar histórico</span>
    <span className="history-cleanup-arrow">⌄</span>
  </summary>

  <div className="history-cleanup-content">
    <p>
      Remova transações antigas sem perder o saldo e os investimentos
      acumulados.
    </p>

    <label htmlFor="cleanup-period">
      Período do histórico
    </label>

    <select
      id="cleanup-period"
      value={cleanupPeriod}
      onChange={(event) => setCleanupPeriod(event.target.value)}
      disabled={cleaningHistory || countingCleanup}
    >
      <option value="1">Mais de 1 mês</option>
      <option value="3">Mais de 3 meses</option>
      <option value="6">Mais de 6 meses</option>
      <option value="12">Mais de 12 meses</option>
      <option value="todos">Todo o histórico</option>
    </select>

    <p aria-live="polite">
      {countingCleanup
        ? "Calculando transações..."
        : cleanupCountError
          ? cleanupCountError
          : cleanupCount === null
            ? "Aguardando dados financeiros..."
            : `${cleanupCount} transação(ões) serão removidas.`}
    </p>

    <button
      type="button"
      className="history-cleanup-btn"
      disabled={
        cleaningHistory ||
        countingCleanup ||
        !financialDataLoaded ||
        cleanupCount === null ||
        cleanupCountError !== "" ||
        cleanupCount === 0
      }
      onClick={() =>
        handleCleanTransactionHistory(
          cleanupPeriod === "todos"
            ? null
            : Number(cleanupPeriod)
        )
      }
    >
      {cleaningHistory ? "Limpando..." : "Limpar histórico"}
    </button>

    <small>
      A operação não pode ser desfeita pela interface. Confira a
      seleção antes de confirmar.
    </small>
  </div>
</details>
          </div>

          {/* Dashboard */}
          <Home
      transactions={filteredTransactions}
      allTransactions={transactions}
      loans={loans}
      scheduledBills={scheduledBills}
      goal={goal}
      limiteGastos={limiteGastos}
      onSalvarLimiteGastos={handleSaveSpendingLimit}
      onUsarValorGuardado={handleUseSavedMoney}
      onEditarMeta={handleEditGoalFromDashboard}
      saldoBase={saldoBase}
      investimentosBase={investimentosBase}
      financialDataLoaded={financialDataLoaded}
      valorGuardadoUtilizadoBase={valorGuardadoUtilizadoBase}
      preservedChartHistory={preservedChartHistory}
    />
          {/* Filtros */}
          <div className="filter-toggle">
            <button
              onClick={() => setShowFilters(!showFilters)}
            >
              {showFilters
                ? "🔽 Esconder Filtros"
                : "▶ Mostrar Filtros"}
            </button>
          </div>

          {showFilters && (
            <FilterBar
              filtro={filtro}
              setFiltro={setFiltro}
            />
          )}

          {/* Transações + Painel lateral */}
          <div className="transactions-container">
            {/* Últimas Transações */}
            <div className="panel-card">
              <div className="panel-header">
                <h3>Últimas Transações</h3>

                {filteredTransactions.length >= 2 && (
                  <button
                    className="loan-toggle"
                    onClick={() =>
                      setShowTransactions(!showTransactions)
                    }
                    title={
                      showTransactions
                        ? "Recolher lista"
                        : "Expandir lista"
                    }
                  >
                    {showTransactions ? "Fechar" : "Abrir"}
                  </button>
                )}
              </div>

              {(showTransactions ||
                filteredTransactions.length < 2) && (
                <TransactionList
                  transactions={filteredTransactions}
                  deleteTransaction={removeTransaction}
                  editTransaction={editTransaction}
                />
              )}
            </div>

            <div className="side-panel">
              {/* Empréstimos */}
              <div className="panel-card">
                <div className="panel-header">
                  <h3>Empréstimos</h3>

                  <div className="loan-actions">
                    <button
                      className="loan-add"
                      onClick={() => {
                        setEditingLoan(null);
                        setLoanModalOpen(true);
                      }}
                      title="Adicionar empréstimo"
                    >
                      +
                    </button>

                    {loans.length >= 2 && (
                      <button
                        className="loan-toggle"
                        onClick={() => setShowLoans(!showLoans)}
                        title={
                          showLoans
                            ? "Recolher lista"
                            : "Expandir lista"
                        }
                      >
                        {showLoans ? "Fechar" : "Abrir"}

function formatLocalDate(date) {
  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, "0");
  const day = String(date.getDate()).padStart(2, "0");

  return `${year}-${month}-${day}`;
}

function getDiasAteVirarOMes() {
  const hoje = new Date();

  const inicioProximoMes = new Date(
    hoje.getFullYear(),
    hoje.getMonth() + 1,
    1
  );

  const hojeSemHorario = new Date(
    hoje.getFullYear(),
    hoje.getMonth(),
    hoje.getDate()
  );

  return Math.round(
    (inicioProximoMes - hojeSemHorario) / (1000 * 60 * 60 * 24)
  );
}

function getDateThreeMonthsAgo() {
  const today = new Date();
  const date = new Date(today);

  date.setDate(1);
  date.setMonth(date.getMonth() - 3);

  const lastDay = new Date(
    date.getFullYear(),
    date.getMonth() + 1,
    0
  ).getDate();

  date.setDate(Math.min(today.getDate(), lastDay));

  return date;
}

import { useEffect, useState } from "react";

function AnimatedNumber({

    value,
    duration = 900,

}){

    const [displayValue, setDisplayValue] = useState(0);

    useEffect(()=>{

        let start = 0;

        const increment = value / (duration / 16);

        const timer = setInterval(()=>{

            start += increment;

            if(start >= value){

                start = value;

                clearInterval(timer);

            }

            setDisplayValue(start);

        },16);

        return ()=>clearInterval(timer);

    },[value,duration]);

    return (

        displayValue.toLocaleString(

            "pt-BR",

            {

                style:"currency",

                currency:"BRL",

            }

        )

    );

}

export default AnimatedNumber;

import { useEffect, useState } from "react";
import { supabase } from "../services/supabase";
import "../styles/header.css";

const AVATAR_BUCKET = "avatars";
const SIGNED_URL_SECONDS = 60 * 60;

function Header({ onOpenProfile, onGoHome, currentPage = "home" }) {
  const [avatarUrl, setAvatarUrl] = useState("");
  const [userId, setUserId] = useState("");
  const [accessibilityOpen, setAccessibilityOpen] = useState(false);

  useEffect(() => {
    let cancelled = false;

    async function loadAvatar() {
      const { data, error } = await supabase.auth.getUser();

      if (error || !data.user) {
        if (!cancelled) {
          setAvatarUrl("");
          setUserId("");
        }
        return;
      }

      const user = data.user;
      const avatarPath = user.user_metadata?.avatar_path;

      if (cancelled) return;

      setUserId(user.id);

      if (!avatarPath) {
        setAvatarUrl("");
        return;
      }

      const { data: signedData, error: signedError } =
        await supabase.storage
          .from(AVATAR_BUCKET)
          .createSignedUrl(avatarPath, SIGNED_URL_SECONDS);

      if (cancelled) return;

      if (signedError) {
        console.error("Erro ao carregar foto do perfil:", signedError);
        setAvatarUrl("");
        return;
      }

      setAvatarUrl(signedData?.signedUrl || "");
    }

    loadAvatar();

    const {
      data: { subscription },
    } = supabase.auth.onAuthStateChange((event, session) => {
      if (event === "SIGNED_OUT" || !session?.user) {
        setAvatarUrl("");
        setUserId("");
      } else if (
        event === "SIGNED_IN" ||
        event === "USER_UPDATED"
      ) {
        loadAvatar();
      }
    });

    return () => {
      cancelled = true;
      subscription.unsubscribe();
    };
  }, []);

  async function logout() {
    const { error } = await supabase.auth.signOut();

    if (error) {
      console.error("Erro ao sair:", error);
    }
  }

  return (
    <header className="header">
      <button

function ModalWizard({
  user,
  transactions,
  setTransactions,
  onClose,
}) {

  const [step, setStep] = useState(1);

  const [formData, setFormData] = useState({
    tipo: "",
    categoria: "",
    descricao: "",
    valor: "",
    data: "",
    observacao: "",
  });
  const [categories, setCategories] = useState([]);

  const [pendingTransactions, setPendingTransactions] = useState([]);

  function adicionarMovimentacao() {

    setPendingTransactions((prev) => [

      ...prev,

      {

        ...formData,

        id: crypto.randomUUID(),

        valor: parseCurrency(formData.valor),

        user_id: user.id,

      },

    ]);

    setFormData({

      tipo: "",
      categoria: "",
      descricao: "",
      valor: "",
      data: "",
      observacao: "",

    });

    setStep(1);

  }

  useEffect(() => {

    

    if (!user) return;

    loadCategories();

}, [user]);

   function validarFormulario() {

      if (!formData.categoria) return "Selecione uma categoria.";

      if (!formData.descricao.trim()) return "Informe uma descrição.";

      if (!formData.valor) return "Informe um valor.";

      if (!formData.data) return "Informe uma data.";

      return null;

    }

async function loadCategories() {

  try {

      const data = await getCategories(user.id);

      setCategories(data);

  } catch (err) {

      console.error(err);

  }

}

async function addCategory(nome, tipo) {

  try {

      await createCategory({

          nome,
          tipo,
          prioridade: false,
          user_id: user.id,

      });

      await loadCategories();

  } catch (err) {

      console.error(err);

  }

}

async function editarCategoria(category) {
  const novoNome = prompt(
      "Novo nome da categoria:",
      category.nome
  );

  if (!novoNome || !novoNome.trim()) {
      return;
  }

  try {
      await updateCategory(category.id, {
          nome: novoNome.trim(),
      });

      await loadCategories();

  } catch (err) {
      console.error("Erro ao editar categoria:", err);
      alert("Erro ao editar categoria.");
  }
}

async function removerCategoria(category) {
  const confirmar = window.confirm(
      `Excluir a categoria "${category.nome}"?`
  );

  if (!confirmar) return;

  try {
      await deleteCategory(category.id);

      if (formData.categoria === category.nome) {
          setFormData((prev) => ({
              ...prev,
              categoria: "",
          }));
      }

      await loadCategories();

  } catch (err) {
      console.error("Erro ao excluir categoria:", err);
      alert("Erro ao excluir categoria.");
  }
}

async function confirmarTudo() {

  const erro = validarFormulario();

  if (erro) {

      alert(erro);

      return;

  }

  const todas = [...pendingTransactions];

  todas.push({

      ...formData,

      id: crypto.randomUUID(),

      valor: parseCurrency(formData.valor),

      user_id: user.id,

  });

  try {

      await createTransactions(todas);

      const data = await getTransactions(user.id);

      setTransactions(data);

      onClose();

  } catch (err) {

      console.error(err);

      alert("Erro ao salvar as movimentações.");

  }

}

  console.log("Categorias do Modal:", categories);

  return (

    <div>

      <h2>Nova Movimentação</h2>

      {step === 1 && (

        <WizardStepType

          setFormData={setFormData}
          setStep={setStep}
        />

      )}

      {step === 2 && (

      <WizardStepInfo
      formData={formData}
      setFormData={setFormData}
      categories={categories}
      addCategory={addCategory}
      editarCategoria={editarCategoria}
      removerCategoria={removerCategoria}

          adicionarMovimentacao={adicionarMovimentacao}
          confirmarTudo={confirmarTudo}
        />

      )}

      <div className="wizard-buttons">

        {step > 1 && (

          <button
            onClick={() => setStep(step - 1)}
          >
            ⬅ Voltar
          </button>

        )}

        {step > 1 && step < 3 && (

    <button
    onClick={() => {

        if(step === 2){

            if(!formData.categoria){

                alert("Selecione uma categoria.");

                return;

            }

            if(!formData.descricao.trim()){

                alert("Informe uma descrição.");

    >

import { useState } from "react";
import CurrencyInput from "react-currency-input-field";

function WizardStepInfo({
    formData,
    setFormData,
    categories,
    addCategory,
    editarCategoria,
    removerCategoria,
}) {
    const categorias = {
        entrada: [
            "Salário",
            "Freelance",
            "Venda",
            "Presente",
            "Outros",
        ],

        saida: [
            "Mercado",
            "Moradia",
            "Transporte",
            "Saúde",
            "Lazer",
            "Educação",
            "Internet",
            "Outros",
        ],

        investimento: [
            "Tesouro",
            "CDB",
            "Ações",
            "ETF",
            "FII",
            "Bitcoin",
        ],

        renda_passiva: [
            "Dividendos",
            "Aluguel",
            "Juros",
            "Royalties",
        ],
    };

    const [novaCategoria, setNovaCategoria] = useState("");
    const [criandoCategoria, setCriandoCategoria] = useState(false);

    const categoriasPadrao =
        categorias[formData.tipo] || [];

    const categoriasUsuario = categories.filter(
        (c) => c.tipo === formData.tipo
    );

    const categoriaUsuarioSelecionada =
        categoriasUsuario.find(
            (c) => c.nome === formData.categoria
        );

    function selecionarCategoriaPadrao(valor) {
        setFormData((prev) => ({
            ...prev,
            categoria: valor,
        }));

        setCriandoCategoria(false);
    }

    function selecionarCategoriaUsuario(e) {
        const valor = e.target.value;

        if (valor === "__criar__") {
            setCriandoCategoria(true);

            setFormData((prev) => ({
                ...prev,
                categoria: "",
            }));

            return;
        }

        setCriandoCategoria(false);

        setFormData((prev) => ({
            ...prev,
            categoria: valor,
        }));
    }

    async function criarNovaCategoria() {
        const nome = novaCategoria.trim();

        if (!nome) {
            return;
        }

        try {
            await addCategory(nome, formData.tipo);

            setFormData((prev) => ({
                ...prev,
                categoria: nome,
            }));

            setNovaCategoria("");
            setCriandoCategoria(false);
        } catch (err) {
            console.error(
                "Erro ao criar categoria:",
                err
            );
        }
    }

    return (
        <div>

            <h2>
                {formData.tipo === "entrada" &&
                    "Nova Entrada"}

                {formData.tipo === "saida" &&
                    "Nova Saída"}

                {formData.tipo === "investimento" &&
                    "Novo Investimento"}

                {formData.tipo === "renda_passiva" &&
                    "Nova Renda Passiva"}
            </h2>

            <div className="category-section">

                {/* CATEGORIAS PADRÃO */}

                <span className="category-title">
                    Categorias padrão
                </span>

                <select
                    className="category-select"
                    value={
                        categoriasPadrao.includes(
                            formData.categoria
                        )
                            ? formData.categoria
                            : ""
                    }
                    onChange={(e) =>
                        selecionarCategoriaPadrao(
                            e.target.value
                        )
                    }
                >
                    <option value="">
                        Categoria
                    </option>

                    {categoriasPadrao.map(
                        (categoria) => (
                            <option
                                key={`padrao-${categoria}`}
                                value={categoria}
                            >
                                {categoria}
                            </option>
                        )
                    )}
                </select>


                {/* MINHAS CATEGORIAS */}

                <span className="category-title">
                    Minhas categorias
                </span>

                <div className="custom-category-select">

                    <select
                        className="category-select"
                        value={
                            criandoCategoria
                                ? "__criar__"
                                : categoriaUsuarioSelecionada
                                    ? formData.categoria
                                    : ""
                        }
                        onChange={
                            selecionarCategoriaUsuario
                        }
                    >

                        <option value="">
                            Categoria
                        </option>

                        <option value="__criar__">
                            ＋ Criar categoria
                        </option>

                        {categoriasUsuario.map(
                            (categoria) => (
                                <option
                                    key={categoria.id}
                                    value={categoria.nome}
                                >
                                    {categoria.nome}
                                </option>
                            )
                        )}

                    </select>


                    {/* AÇÕES DA CATEGORIA */}

                    {categoriaUsuarioSelecionada &&
                        !criandoCategoria && (
                            <div className="category-actions">

                                <button
                                    type="button"
                                    onClick={() =>
                                        editarCategoria(
                                            categoriaUsuarioSelecionada
                                        )
                                    }
                                    title="Editar"
                                >
                                    ✏️
                                </button>

                                <button
                                    type="button"
                                    onClick={() =>
                                        removerCategoria(
                                            categoriaUsuarioSelecionada
                                        )
                                    }
                                    title="Excluir"
                                >
                                    🗑️
                                </button>

                            </div>
                        )}

                </div>


                {/* CRIAR CATEGORIA */}

                {criandoCategoria && (

                            }
                        >
                            Adicionar
                        </button>

                    </div>
                )}

            </div>


            {/* DESCRIÇÃO */}

            <input
                placeholder="Descrição *"
                value={formData.descricao}
                onChange={(e) =>
                    setFormData({
                        ...formData,
                        descricao:
                            e.target.value,

                    setFormData({
                        ...formData,
                        observacao:
                            e.target.value,
                    })
                }
            />

        </div>
    );
}

import { useEffect, useRef, useState } from "react";
import { supabase } from "../services/supabase";
import "../styles/profile.css";

const AVATAR_BUCKET = "avatars";
const MAX_AVATAR_SIZE = 2 * 1024 * 1024;
const SIGNED_URL_SECONDS = 60 * 60;

function Profile() {
  const fileInputRef = useRef(null);

  const [user, setUser] = useState(null);
  const [nome, setNome] = useState("");
  const [avatarUrl, setAvatarUrl] = useState("");
  const [avatarPath, setAvatarPath] = useState("");
  const [tema, setTema] = useState("dark");
  const [idioma, setIdioma] = useState("pt-BR");
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [uploading, setUploading] = useState(false);
  const [message, setMessage] = useState("");
  const [errorMessage, setErrorMessage] = useState("");

  useEffect(() => {
    let cancelled = false;

    async function loadProfile() {
      try {
        const { data, error } = await supabase.auth.getUser();

        if (error) throw error;

        if (!data.user) {
          if (!cancelled) {
            setErrorMessage("Não foi possível carregar sua conta.");
          }
          return;
        }

        const currentUser = data.user;
        const metadata = currentUser.user_metadata || {};
        const savedPath = metadata.avatar_path || "";

        let signedUrl = "";

        if (savedPath) {
          const { data: signedData, error: signedError } =
            await supabase.storage
              .from(AVATAR_BUCKET)
              .createSignedUrl(savedPath, SIGNED_URL_SECONDS);

          if (!signedError) {
            signedUrl = signedData?.signedUrl || "";
          } else {
            console.error(
              "Não foi possível carregar a foto:",
              signedError.message
            );
          }
        } else if (metadata.avatar_url) {
          // Compatibilidade com perfis antigos que usam URL pública.
          signedUrl = metadata.avatar_url;
        }

        let savedTheme = "dark";
        let savedLanguage = "pt-BR";

        try {
          const storedTheme = localStorage.getItem("atlas-theme");
          const storedLanguage = localStorage.getItem("atlas-language");

          if (["dark", "light", "system"].includes(storedTheme)) {
            savedTheme = storedTheme;
          }

          if (["pt-BR", "en", "es"].includes(storedLanguage)) {
            savedLanguage = storedLanguage;
          }
        } catch {
          // As preferências padrão continuam disponíveis.
        }

        if (cancelled) return;

        setUser(currentUser);
        setNome(metadata.full_name || metadata.name || "");
        setAvatarPath(savedPath);
        setAvatarUrl(signedUrl);
        setTema(savedTheme);
        setIdioma(savedLanguage);
      } catch (error) {
        if (!cancelled) {
          setErrorMessage(
            error.message || "Não foi possível carregar sua conta."
          );
        }
      } finally {
        if (!cancelled) setLoading(false);
      }
    }

    loadProfile();

    return () => {
      cancelled = true;
    };
  }, []);

  function clearMessages() {
    setMessage("");
    setErrorMessage("");
  }

  async function saveProfile(event) {
    event.preventDefault();
    clearMessages();

    if (!user) return;

    if (!nome.trim()) {
      setErrorMessage("Informe seu nome.");
      return;
    }

    setSaving(true);

    try {
      const { data, error } = await supabase.auth.updateUser({
        data: {
          ...user.user_metadata,
          full_name: nome.trim(),
          avatar_path: avatarPath || null,
          // Não gravamos a URL assinada: ela expira.
          avatar_url: null,
        },
      });

      if (error) throw error;

      setUser(data.user);
      setMessage("Informações do perfil salvas com sucesso.");
    } catch (error) {
      setErrorMessage(
        `Não foi possível salvar: ${error.message}`
      );
    } finally {
      setSaving(false);
    }
  }

  async function uploadAvatar(event) {
    const file = event.target.files?.[0];

    event.target.value = "";

    if (!file || !user) return;

    clearMessages();

    const allowedTypes = [
      "image/jpeg",
      "image/png",
      "image/webp",
    ];

    if (!allowedTypes.includes(file.type)) {
      setErrorMessage("Selecione uma imagem JPG, PNG ou WEBP.");
      return;
    }

    if (file.size > MAX_AVATAR_SIZE) {
      setErrorMessage("A imagem deve ter no máximo 2 MB.");
      return;
    }

    setUploading(true);

    const filePath = `${user.id}/avatar`;

    try {
      const { error: uploadError } = await supabase.storage
        .from(AVATAR_BUCKET)
        .upload(filePath, file, {
          upsert: true,
          contentType: file.type,
          cacheControl: "3600",
        });

      if (uploadError) throw uploadError;

      const { data: signedData, error: signedError } =
        await supabase.storage
          .from(AVATAR_BUCKET)
          .createSignedUrl(filePath, SIGNED_URL_SECONDS);

      if (signedError) throw signedError;

      const { data: updatedData, error: updateError } =
        await supabase.auth.updateUser({
          data: {
            ...user.user_metadata,
            full_name: nome.trim(),
            avatar_path: filePath,
            avatar_url: null,
          },
        });

      if (updateError) throw updateError;

      setUser(updatedData.user);
      setAvatarPath(filePath);
      setAvatarUrl(signedData.signedUrl);
      setMessage("Foto de perfil atualizada.");
    } catch (error) {
      setErrorMessage(
        error.message ||
          "Não foi possível enviar a foto. Verifique o Storage e suas políticas."
      );
    } finally {
      setUploading(false);
    }
  }

  async function removeAvatar() {
    clearMessages();

    if (!user || saving || uploading) return;

    setSaving(true);

    try {
      if (avatarPath) {
        const { error: deleteError } = await supabase.storage
          .from(AVATAR_BUCKET)
          .remove([avatarPath]);

        if (deleteError) throw deleteError;
      }

      const { data, error } = await supabase.auth.updateUser({
        data: {
          ...user.user_metadata,
          full_name: nome.trim(),
          avatar_path: null,
          avatar_url: null,
        },
      });

      if (error) throw error;

      setUser(data.user);
      setAvatarPath("");
      setAvatarUrl("");
      setMessage("Foto personalizada removida.");
    } catch (error) {
      setErrorMessage(
        `Não foi possível remover a foto: ${error.message}`
      );
    } finally {
      setSaving(false);
    }
  }

  function savePreferences(nextTheme = tema, nextLanguage = idioma) {
    clearMessages();

    try {
      localStorage.setItem("atlas-theme", nextTheme);
      localStorage.setItem("atlas-language", nextLanguage);

      setTema(nextTheme);
      setIdioma(nextLanguage);

      setMessage("Preferências salvas neste navegador.");
    } catch {
      setErrorMessage("Não foi possível salvar as preferências.");
    }
  }

  async function resetPassword() {
    clearMessages();

    if (!user?.email) {
      setErrorMessage("Não encontramos um e-mail para esta conta.");
      return;
    }

    try {
      const { error } = await supabase.auth.resetPasswordForEmail(
        user.email,
        {
          redirectTo: `${window.location.origin}/`,
        }
      );

      if (error) throw error;

      setMessage(
        "Se o serviço estiver configurado, você receberá um e-mail para redefinir sua senha."
      );
    } catch (error) {
      setErrorMessage(
        `Não foi possível enviar o e-mail: ${error.message}`
      );
    }
  }

  if (loading) {
    return (
      <main className="profile-page">
        <div className="profile-loading">Carregando perfil...</div>
      </main>
    );
  }

  if (!user) {
    return (
      <main className="profile-page">
        <div className="profile-loading">
          {errorMessage || "Faça login para acessar seu perfil."}
        </div>
      </main>
    );
  }

  return (
    <main className="profile-page">
      <div className="profile-heading">
        <h2>Perfil e Configurações</h2>
        <p>
          Gerencie suas informações e preferências do Atlas Finance.
        </p>
      </div>

      {message && (
        <div className="profile-message success" role="status">
          {message}
        </div>
      )}

      {errorMessage && (
        <div className="profile-message error" role="alert">
          {errorMessage}
        </div>
      )}

      <section className="profile-card avatar-card">
        <div className="profile-avatar">
          {avatarUrl ? (
            <img src={avatarUrl} alt="Foto de perfil" />
          ) : (
            <div className="profile-avatar-placeholder">
              <span>💜</span>
            </div>
          )}
        </div>

        <div className="avatar-content">
          <h3>Foto do perfil</h3>
          <p>Esta foto será exibida no seu perfil.</p>

          <div className="avatar-actions">
            <button
              className="profile-primary-button"
              type="button"
              onClick={() => fileInputRef.current?.click()}
              disabled={uploading || saving}
            >
              {uploading ? "Enviando..." : "↑ Alterar foto"}
            </button>

            <button
              className="profile-secondary-button"
              type="button"
              onClick={removeAvatar}
              disabled={!avatarPath || saving || uploading}
            >
              Remover foto
            </button>

            <input
              ref={fileInputRef}
              type="file"
              accept="image/jpeg,image/png,image/webp"
              onChange={uploadAvatar}
              hidden
            />
          </div>

          <p className="profile-help">
            Formatos aceitos: JPG, PNG ou WEBP. Tamanho máximo: 2 MB.
          </p>
        </div>
      </section>

      <form onSubmit={saveProfile}>
        <section className="profile-card">
          <div className="profile-section-heading">
            <h3>Informações pessoais</h3>
            <p>Seus dados básicos de identificação.</p>
          </div>

          <div className="profile-fields">
            <label className="profile-field">
              <span>Nome</span>
              <input
                type="text"
                value={nome}
                onChange={(event) => setNome(event.target.value)}
                placeholder="Seu nome"
                maxLength={80}
                required
              />
            </label>

            <label className="profile-field">
              <span>E-mail</span>
              <input
                type="email"
                value={user.email || ""}
                readOnly
                disabled
              />
              <small>
                O e-mail está vinculado à sua conta de autenticação.
              </small>
            </label>
          </div>

          <div className="profile-form-actions">
            <button
              className="profile-primary-button"
              type="submit"
              disabled={saving || uploading}
            >
              {saving ? "Salvando..." : "✓ Salvar alterações"}
            </button>
          </div>
        </section>
      </form>

      <section className="profile-card">
        <div className="profile-section-heading">
          <h3>Preferências</h3>
          <p>Personalize sua experiência no Atlas Finance.</p>
        </div>

        <div className="profile-preference">
          <div>
            <strong>Tema</strong>
            <p>Escolha a aparência da aplicação.</p>
          </div>

          <div className="profile-segmented-control">
            {[
              { value: "dark", label: "Escuro" },
              { value: "light", label: "Claro" },
              { value: "system", label: "Sistema" },
            ].map((option) => (
              <button
                key={option.value}
                type="button"
                className={tema === option.value ? "active" : ""}
                aria-pressed={tema === option.value}
                onClick={() => savePreferences(option.value, idioma)}
              >
                {option.label}
              </button>
            ))}
          </div>
        </div>

        <div className="profile-preference">
          <div>
            <strong>Idioma</strong>
            <p>Selecione o idioma de sua preferência.</p>
          </div>

          <select
            value={idioma}
            onChange={(event) =>
              savePreferences(tema, event.target.value)
            }
            aria-label="Idioma da aplicação"
          >
            <option value="pt-BR">Português (Brasil)</option>
            <option value="en">English</option>
            <option value="es">Español</option>
          </select>
        </div>
      </section>

      <section className="profile-card">
        <div className="profile-section-heading">
          <h3>Segurança</h3>
          <p>Gerencie o acesso à sua conta.</p>
        </div>

        <div className="profile-security-row">
          <div>
            <strong>Alterar senha</strong>
            <p>Receba um e-mail para redefinir sua senha.</p>
          </div>

          <button
            className="profile-secondary-button"
            type="button"
            onClick={resetPassword}
          >
            ↗ Redefinir senha
          </button>
        </div>
      </section>
    </main>
  );
}

export default Profile;
